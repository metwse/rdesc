52static inline int resize_stack(struct rdesc_stack **s, size_t cap)
53{
54 struct rdesc_stack *new =
55 xrealloc(*s, sizeof(struct rdesc_stack) + cap * (*s)->element_size);

59 (*s)->cap = cap;

61 return 0;

67/* returns non-zero value if resize failure */
68static int stack_reserve(struct rdesc_stack **s, size_t reserved_space)
69{
70 size_t increased_cap = (*s)->cap;
71
72 while (increased_cap <= (*s)->len + reserved_space) {
73 if (increased_cap > STACK_MAX_CAP / (2 * (*s)->element_size))
74 return 1;
75
76 increased_cap *= 2;
77 }
78
79 if (increased_cap != (*s)->cap)
80 return resize_stack(s, increased_cap);
81
82 return 0;
83}

122void *rdesc_stack_multipush(struct rdesc_stack **s, void *element, size_t count)
123{
124 /* return null if grow failed */
125 if (stack_reserve(s, count) || (xmultipush(count)))
126 return NULL;
127
128 void *top = elem_at(*s, (*s)->len);
129
130 if (element)
131 memcpy(top, element, (*s)->element_size * count);
132 (*s)->len += count;
133
134 return top;
135}

142void *rdesc_stack_multipop(struct rdesc_stack **s, size_t count)
143{
144 runtime_assertion((*s)->len >= count, "stack underflow");
145
146 size_t decreased_cap = (*s)->cap;
147
148 while ((*s)->len * 4 <= decreased_cap &&
149 decreased_cap >= STACK_INITIAL_CAP * 2)
150 decreased_cap /= 2;
151
152 (*s)->len -= count;
153
154 if (decreased_cap != (*s)->cap)
155 resize_stack(s, decreased_cap);
156
157 return elem_at(*s, (*s)->len);
158}
