17/* We only call this function with nonterminals, starting from <stmt>. */
18double pm_interpreter(struct rdesc_node n)
19{
20 switch (rid(n)) {
21 case NT_STMT:
22 /* continue evaluation from first child (<expr>) */
23 return pm_interpreter(rchild(n, 0));
24
25 case NT_EXPR:
26 switch (ralt_idx(n)) {
27 case 0: /* <exponentiation_expr> "|" <pipe_expr>*/
28 /* flip <pipe_expr> to left-associative order */
29 rdesc_flip_left(n, 2);
30
31 /* Start pipe evaluation from left-to-right. */
32 return pm_interpret_pipe(rchild(n, 2),
33 pm_interpreter(rchild(n, 0)));
34 default: /* <exponentiation_expr> */
35 return pm_interpreter(rchild(n, 0));
36 }
37
38 case NT_EXPONENTIATION_EXPR:
39 /* <num> <exponentiation_expr_rest> */
40 return pow(pm_extract_num(rchild(n, 0)),
41 pm_interpreter(rchild(n, 1)));
42
43 case NT_EXPONENTIATION_EXPR_REST:
44 switch (ralt_idx(n)) {
45 case 0: /* "^" <num> <exponentiation_expr_rest> */
46 return pow(pm_extract_num(rchild(n, 1)),
47 pm_interpreter(rchild(n, 2)));
48 default: /* E */
49 return 1;
50 }
51 }
52
53 return 0; // GCOVR_EXCL_LINE: Unreachable
54}

59double pm_extract_num(struct rdesc_node num)
60{
61 char *seminfo;
62
63 /* seminfo stores a pointer to number, so it is a char **.
64 *
65 * Aside: We need to use memcpy to retrieve char * from char ** to not
66 * break strict-aliasing rule. */
67 memcpy(&seminfo, rseminfo(num), sizeof(char *));
68
69 double seminfo_num = atof(seminfo);
70
71 free(seminfo);
72
73 return seminfo_num;
74}

81 double lhs /* value at left-hand side of pipe */)
82{

85 return pm_interpret_function(rchild(pipe, 2),

87
88 default: /* <function_call> */
89 return pm_interpret_function(rchild(pipe, 0), lhs);

94static void collect_function_args(struct rdesc_node args,
95 size_t *argc, double **argv)
96{
97 size_t current_idx;
98
99 switch (rid(args)) {
100 case NT_FUNCTION_ARG_LS: /* <expr> <function_arg_ls_rest> */
101 current_idx = (*argc)++;
102
103 collect_function_args(rchild(args, 1), argc, argv);
104
105 break;
106
107 case NT_FUNCTION_ARG_LS_REST:
108 switch (ralt_idx(args)) {
109 case 0: /* "," <expr> <function_arg_ls_rest> */
110 current_idx = (*argc)++;
111
112 collect_function_args(rchild(args, 2), argc, argv);
113
114 break;
115
116 case 1: /* E */
117 assert(*argv = malloc(sizeof(double) * *argc));
118
119 return;
120 }
121 }
122
123 (*argv)[current_idx] =
124 pm_interpreter(rchild(args, rid(args) == NT_FUNCTION_ARG_LS ? 0 : 1));
125}

127double pm_interpret_function(struct rdesc_node function,
128 double lhs)
129{
130 char *function_name;
131
132 /* child at index 0 is always function name (identifier) */
133 memcpy(&function_name, rseminfo(rchild(function, 0)), sizeof(char *));
134
135 size_t function_id;
136
137 for (function_id = 0; function_id < pm_function_count; function_id++) {
138 if (strcmp(function_name, pm_function_names[function_id]) == 0)
139 break;
140 }
141
142 size_t argc = 0;
143 double *argv = NULL;
144 double result;
145
146 if (ralt_idx(function) == 0) { /* ident "(" <function_arg_ls> ")" */
147 collect_function_args(rchild(function, 2), &argc, &argv);
148 } else { /* ident */ }
149
150 if (function_id == pm_function_count) {
151 fprintf(stderr, "Unknown function %s, ignoring\n", function_name);
152
153 result = lhs;
154 } else {
155 result = pm_functions[function_id](lhs, argc, argv);
156 }
157
158 if (argv)
159 free(argv);
160
161 free(function_name);
162
163 return result;
164}
